#! label: "get-data"
treatment_data <- read.csv("data/Clinical_Treatment_Data.csv")
all_meta_columns <- colnames(treatment_data)
drug_columns <- all_meta_columns[grepl("^Drug",all_meta_columns)]
treatment_data$cancer_subtype <- case_when(((treatment_data$er_status_by_ihc == "Positive") | (treatment_data$pr_status_by_ihc == "Positive")) & (treatment_data$her2_status_by_ihc == "Positive") ~ "HR+/HER2+",
                                        ((treatment_data$er_status_by_ihc == "Positive") | (treatment_data$pr_status_by_ihc == "Positive")) & (treatment_data$her2_status_by_ihc == "Negative") ~ "HR+/HER2-",
                                        treatment_data$her2_status_by_ihc == "Positive" ~ "HER2+(HR-)",
                                        (treatment_data$er_status_by_ihc == "Negative") & (treatment_data$pr_status_by_ihc == "Negative") & (treatment_data$her2_status_by_ihc == "Negative") ~ "TNBC",
                                        .default = "Unknown")
